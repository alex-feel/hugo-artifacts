{{/* Hugo PWA module -- precache-list helper. Builds a [{url, revision}] array suitable for Workbox's precacheAndRoute. The whole list is gated on params.pwa.sw.precache.enabled (master switch); when false the SW precaches nothing. CRITICAL invariant: every entry MUST correspond to a resource Hugo actually emits. Workbox's precacheAndRoute install is atomic -- a single 404 fails the whole SW install, so the SW silently never activates. Two mechanisms keep the URLs honest on a subpath (baseURL with a path) deploy, and every entry uses exactly one of them: - Entries that name a page or output format (site.Home, the offline page via site.GetPage, the WebAppManifest output, the recent-pages list) are derived from the real rendered resource, through pwa/lib/resource-url.html. NOT through .RelPermalink directly: that method drops the baseURL's path when canonifyURLs is true, because Hugo's canonification step puts it back afterwards -- and that step never runs over sw.js, which is not HTML. - Entries that come from a raw config path (the offline fallback image, the rfg-static manifest href, and every extra_urls value) are normalized through pwa/lib/absolute-url.html, which carries the baseURL's path onto a value written with a leading slash. Without it, a consumer's "/about/" would be precached as "/about/" on a site published under "/docs/", the fetch would 404, and the atomic install would fail -- taking down the whole service worker, not just that one entry. Sources of entries (each opt-in via params): - Homepage when params.pwa.sw.precache.include_homepage is true - Offline page when BOTH params.pwa.sw.precache.include_offline_page AND params.pwa.sw.offline.enabled are true (the page exists only then -- it is created by content/_content.gotmpl under the same condition) - Offline fallback image when offline is enabled and params.pwa.sw.offline.fallback_image is set - Manifest when params.pwa.sw.precache.include_manifest is true AND the manifest actually renders (templated mode requires webappmanifest in outputs.home; rfg-static mode points at the consumer's static file) - Top-N regular pages by Lastmod (params.pwa.sw.precache.include_recent_pages) - Any extra URLs in params.pwa.sw.precache.extra_urls Pages whose URL contains an exclude_globs token (after stripping the "*") are dropped. The assembled list is deduplicated by url, first occurrence winning, before it is returned. Workbox rejects a precache list that names one URL twice with two different revisions (add-to-cache-list-conflicting-entries), which fails the same atomic install a 404 does; a consumer naming a recent page in extra_urls produces exactly that pair, because the page entry carries a Lastmod revision and the extra_urls entry carries the URL itself. First-occurrence-wins keeps the Lastmod revision, which is the one that actually tracks the file. Returns a slice of dicts which js.Build's params bridge serializes to JSON. */}} {{- $defaults := hugo.Data.pwa.defaults -}} {{- $params := site.Params.pwa | default dict -}} {{- $swParams := $params.sw | default dict -}} {{- $swCfg := merge $defaults.sw $swParams -}} {{- $list := slice -}} {{- if $swCfg.precache.enabled -}} {{/* Homepage (resolved through the rendered resource) */}} {{- if $swCfg.precache.include_homepage -}} {{- with site.Home -}} {{- $list = $list | append (dict "url" (partial "pwa/lib/resource-url.html" .) "revision" (printf "%d" .Lastmod.Unix)) -}} {{- end -}} {{- end -}} {{/* Offline page -- only when offline fallback is enabled (the page exists only then) */}} {{- if and $swCfg.precache.include_offline_page $swCfg.offline.enabled -}} {{- $offlineRef := printf "/%s" (strings.Trim ($swCfg.offline.fallback_url | default "/offline/") "/") -}} {{- with site.GetPage $offlineRef -}} {{- $list = $list | append (dict "url" (partial "pwa/lib/resource-url.html" .) "revision" (printf "%d" .Lastmod.Unix)) -}} {{- end -}} {{- end -}} {{/* Offline fallback image (precached so the SW catch handler can serve it). Normalized to the same value service-worker.html hands the catch handler, so the precached URL and the matchPrecache lookup agree. */}} {{- if and $swCfg.offline.enabled $swCfg.offline.fallback_image -}} {{- $img := partial "pwa/lib/absolute-url.html" (dict "value" $swCfg.offline.fallback_image "key" "params.pwa.sw.offline.fallback_image") -}} {{- with $img -}} {{- $list = $list | append (dict "url" . "revision" .) -}} {{- end -}} {{- end -}} {{/* Manifest -- only when it actually renders, to avoid an atomic-install 404 */}} {{- if $swCfg.precache.include_manifest -}} {{- $manifestParams := $params.manifest | default dict -}} {{- $mode := $manifestParams.mode | default $defaults.manifest.mode -}} {{- if eq $mode "rfg-static" -}} {{- $href := partial "pwa/lib/absolute-url.html" (dict "value" ($manifestParams.href | default "/site.webmanifest") "key" "params.pwa.manifest.href") -}} {{- with $href -}} {{- $list = $list | append (dict "url" . "revision" "manifest") -}} {{- end -}} {{- else -}} {{- with site.Home.OutputFormats.Get "webappmanifest" -}} {{- $list = $list | append (dict "url" (partial "pwa/lib/resource-url.html" .) "revision" (printf "%d" site.Home.Lastmod.Unix)) -}} {{- end -}} {{- end -}} {{- end -}} {{/* Top-N regular pages by Lastmod, descending */}} {{- $n := $swCfg.precache.include_recent_pages | default 0 -}} {{- if gt $n 0 -}} {{- $excludes := $swCfg.precache.exclude_globs -}} {{- $sorted := first $n (sort site.RegularPages "Lastmod" "desc") -}} {{- range $sorted -}} {{- $url := partial "pwa/lib/resource-url.html" . -}} {{- $skip := false -}} {{- range $excludes -}} {{- $token := replaceRE "\\*" "" . -}} {{- if and $token (strings.Contains $url $token) -}} {{- $skip = true -}} {{- end -}} {{- end -}} {{- if not $skip -}} {{- $list = $list | append (dict "url" $url "revision" (printf "%d" .Lastmod.Unix)) -}} {{- end -}} {{- end -}} {{- end -}} {{/* Extra URLs (consumer-supplied) */}} {{- range $swCfg.precache.extra_urls -}} {{- $extra := partial "pwa/lib/absolute-url.html" (dict "value" . "key" "params.pwa.sw.precache.extra_urls[]") -}} {{- with $extra -}} {{- $list = $list | append (dict "url" . "revision" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- $deduped := slice -}} {{- $seen := slice -}} {{- range $list -}} {{- if not (in $seen .url) -}} {{- $seen = $seen | append .url -}} {{- $deduped = $deduped | append . -}} {{- end -}} {{- end -}} {{- return $deduped -}}
