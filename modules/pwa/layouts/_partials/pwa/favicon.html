{{/* Hugo PWA module -- RFG modern (2024 redesign) favicon head snippet. Emits the minimal head snippet from RealFaviconGenerator's "Less is More" 2024 redesign: - - (when include_svg) - The Web App Manifest is emitted separately by manifest.html. Filenames default to the 2024 RFG redesign output (web-app-manifest-{192,512}x*.png). Set params.pwa.favicon.use_legacy_filenames = true to switch to /android-chrome-* filenames (covers consumers who still ship older RFG packages in static/). The icon-list helper honors the same toggle when emitting the manifest icons[] array. Each href is composed as prefix + configured path and then normalized through pwa/lib/absolute-url.html, so it carries the baseURL's path. The shipped defaults start with "/", which a browser resolves against the ORIGIN, so on a site at https://example.org/docs/ an unnormalized href points outside the site and 404s. */}} {{- $defaults := hugo.Data.pwa.defaults.favicon -}} {{- $params := site.Params.pwa.favicon | default dict -}} {{- $cfg := merge $defaults $params -}} {{- $prefix := strings.TrimSuffix "/" ($cfg.prefix | default "") -}} {{- $faviconIco := partial "pwa/lib/absolute-url.html" (dict "value" (printf "%s%s" $prefix $cfg.favicon_ico_path) "key" "params.pwa.favicon.favicon_ico_path") -}} {{- $faviconSvg := partial "pwa/lib/absolute-url.html" (dict "value" (printf "%s%s" $prefix $cfg.favicon_svg_path) "key" "params.pwa.favicon.favicon_svg_path") -}} {{- $appleTouchIcon := partial "pwa/lib/absolute-url.html" (dict "value" (printf "%s%s" $prefix $cfg.apple_touch_icon_path) "key" "params.pwa.favicon.apple_touch_icon_path") -}} {{- $includeSvg := $cfg.include_svg -}} {{- if $includeSvg }} {{- end }}
