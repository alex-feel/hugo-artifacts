{{/* Normalizes ONE image candidate to a stable dict {url, width, height, type, alt, natural}. Pure value chooser: no HTML, no side effects. Context (dict): - .candidate (any) Either a Hugo resource (page-bundle or global), a string (bundle-relative path, site-root path, or absolute/remote URL), or a map carrying such a string under src, url, or image -- the front-matter shape images: [{src: "/img/cover.jpg"}] many themes use. - .page (page) The current page, for the alt fallback chain and for resolving a string candidate as a bundle or global resource. - .variant (string) Optional. "og" (default): a local raster gets the content-aware 1.91:1 crop that og:image wants. "natural": no crop; the source image's own dimensions are kept -- for JSON-LD entity images (Organization logo, ProfilePage mainEntity image, author Person image), where a 1.91:1 crop destroys logos and portraits. Normalization rules: - Local raster resource (jpg/png/webp/gif/...), variant "og": .Fill "1200x630 Smart" then .Permalink (absolute), .Width (1200), .Height (630), .MediaType.Type. This yields real og:image:width/height/type and a content-aware 1.91:1 crop. - Local raster resource, variant "natural": the source resource's own .Permalink/.Width/.Height, no processing. - SVG resource: passed through WITHOUT processing (Hugo cannot raster-process SVG); url + type only, width/height omitted. - String candidate (unless absolute or protocol-relative): resolved as a page-bundle resource first ($page.Resources.GetMatch, so bundle-relative values like "tile.png" work), else as a global resource (resources.Get). A string that resolves to neither is treated as a bare URL/path: absolutized, no processing, NO fabricated dimensions; type inferred from the extension only when unambiguous (.png/.jpg/.jpeg/.webp/.gif/.svg). natural sub-dict: every resolved image carries `natural` -- the un-cropped source values {url, width, height} (dimensions present only for local raster resources) -- so a consumer that needs the native aspect can read it regardless of the variant that shaped the top-level values. alt chain: seo.image_alt -> resource .Params.alt -> resource .Title (when set to something other than the filename) -> params.seo.image_alt -> resolved page title. So og:image:alt / twitter:image:alt are ALWAYS present when an image is. Returns: {url, width, height, type, alt, natural}, or an empty dict when the candidate is unresolvable. ONE terminal return. */}} {{- $candidate := .candidate -}} {{- $page := .page -}} {{- $variant := .variant | default "og" -}} {{- $result := dict -}} {{- $res := "" -}} {{/* Reduce a non-resource candidate to a usable scalar string BEFORE any stringification. printf "%v" on a map renders Go's debug form (map[src:/img/cover.jpg]), which would be absolutized and published into og:image and every JSON-LD image node as a URL that can never resolve -- so a map-shaped candidate is read through its src, url, or image key instead. A candidate (or extracted value) that is a boolean, a slice, or a map carrying none of those keys holds no path at all: it warns once per page and is skipped rather than published. */}} {{- $str := "" -}} {{- if not (reflect.IsResource $candidate) -}} {{- if reflect.IsMap $candidate -}} {{- range slice "src" "url" "image" -}} {{- if eq $str "" -}} {{- $v := index $candidate . -}} {{- if and (ne $v nil) (not (reflect.IsMap $v)) (not (reflect.IsSlice $v)) (ne (printf "%T" $v) "bool") -}} {{- $str = trim (printf "%v" $v) " \t\n\r" -}} {{- end -}} {{- end -}} {{- end -}} {{- else if and (ne $candidate nil) (not (reflect.IsSlice $candidate)) (ne (printf "%T" $candidate) "bool") -}} {{- $str = trim (printf "%v" $candidate) " \t\n\r" -}} {{- end -}} {{- if and (eq $str "") (or (reflect.IsMap $candidate) (reflect.IsSlice $candidate) (eq (printf "%T" $candidate) "bool")) -}} {{- partial "seo/lib/warn.html" (dict "key" (printf "seo:warn:unusable-image-candidate:%s" $page.RelPermalink) "message" (printf "[seo] Page %q lists an image candidate with no usable value; a map-shaped entry is read through its src, url, or image key, and a boolean or nested list holds no path. That candidate is skipped." $page.RelPermalink) ) -}} {{- end -}} {{- end -}} {{/* Resolve the candidate to a resource when possible. */}} {{- if reflect.IsResource $candidate -}} {{- $res = $candidate -}} {{- else if $str -}} {{/* String candidate: a bundle-relative path resolves as a page resource (the more specific scope wins), a project path as a global resource; otherwise it is treated as a bare URL/path below. */}} {{- if not (or (hasPrefix $str "http://") (hasPrefix $str "https://") (hasPrefix $str "//")) -}} {{- with $page.Resources.GetMatch $str -}} {{- $res = . -}} {{- end -}} {{- if not $res -}} {{- with resources.Get $str -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{/* alt chain (shared by the resource and string branches). */}} {{- $alt := "" -}} {{- with $page.Param "seo.image_alt" -}} {{- $alt = . -}} {{- end -}} {{- if $res -}} {{/* Resource branch: distinguish raster from SVG by sub-type. */}} {{- $isImage := eq $res.ResourceType "image" -}} {{- $isSVG := and $isImage (eq $res.MediaType.SubType "svg") -}} {{/* A resource can be media-typed as an image and still carry no image Hugo can decode: a format Hugo has no decoder for (avif, heic), a file whose bytes are not the image its extension claims (a version-control pointer checked out in place of the binary), or a truncated upload. Reading .Width or calling .Fill DECODES, so on any of those it errors -- and an error here would abort the WHOLE consumer build from a page that configured nothing, because images.html auto-picks bundle resources by name. The sub-type list is the fast path for formats known to have no decoder; the probe covers everything else, since only a decode attempt can tell a sound file from a broken one. Either way an undecodable image degrades to the same url-only rendering an SVG gets. */}} {{- $decodable := false -}} {{- if and $isImage (not $isSVG) -}} {{- if in (slice "jpeg" "png" "gif" "tiff" "bmp" "webp") $res.MediaType.SubType -}} {{/* A width of zero counts as undecodable: .Width answers from a header parse and returns zero rather than erroring when the bytes are not the image the extension claims, so the error alone is not enough. */}} {{- $probe := try $res.Width -}} {{- $decodable = and (not $probe.Err) (gt (int ($probe.Value | default 0)) 0) -}} {{- end -}} {{- if not $decodable -}} {{- partial "seo/lib/warn.html" (dict "key" (printf "seo:warn:undecodable-image:%s" $res.RelPermalink) "message" (printf "[seo] Cannot decode image %q (%s); publishing its URL without dimensions or cropping. Re-export it in a format Hugo can process (jpeg, png, gif, tiff, bmp, webp), or check that the file is complete." $res.RelPermalink $res.MediaType.Type) ) -}} {{- end -}} {{- end -}} {{- if not $alt -}} {{- with $res.Params.alt -}} {{- $alt = . -}} {{- else -}} {{/* Use resource .Title only when it is not merely the filename. */}} {{- $rt := $res.Title -}} {{- if and $rt (ne $rt $res.Name) -}} {{- $alt = $rt -}} {{- end -}} {{- end -}} {{- end -}} {{- if $decodable -}} {{/* Local raster: real dimensions. The og variant adds the content-aware 1.91:1 crop; the natural variant keeps the source as-is. The crop is attempted through try for the same reason the probe above exists: decoding succeeded, but a resize can still fail on a malformed image, and a failed crop degrades to the source's own dimensions rather than taking the build down. */}} {{- $natural := dict "url" $res.Permalink "width" $res.Width "height" $res.Height -}} {{- $chosen := $natural -}} {{- if eq $variant "og" -}} {{- $fill := try ($res.Fill "1200x630 Smart") -}} {{- with $fill.Err -}} {{- partial "seo/lib/warn.html" (dict "key" (printf "seo:warn:uncroppable-image:%s" $res.RelPermalink) "message" (printf "[seo] Cannot crop image %q to the 1.91:1 ratio og:image expects; publishing it uncropped at its own dimensions." $res.RelPermalink) ) -}} {{- else -}} {{- $filled := $fill.Value -}} {{- $chosen = dict "url" $filled.Permalink "width" $filled.Width "height" $filled.Height -}} {{- end -}} {{- end -}} {{- $result = dict "url" $chosen.url "width" $chosen.width "height" $chosen.height "type" $res.MediaType.Type "alt" $alt "natural" $natural -}} {{- else -}} {{/* SVG, an undecodable raster, or another non-raster resource: url + type only, no dimensions. */}} {{- $result = dict "url" $res.Permalink "type" $res.MediaType.Type "alt" $alt "natural" (dict "url" $res.Permalink) -}} {{- end -}} {{- else if $str -}} {{/* String branch: bare URL or site-root path, absolutized, no dimensions. Absolutized through seo/lib/absolute-url.html rather than by piping into absURL, because a site-root path -- the form this branch exists to accept, and the form a static/ file must be written as -- would otherwise resolve against the protocol and host alone and drop the baseURL path on a subpath deployment. This value feeds og:image, og:image:secure_url, twitter:image and every JSON-LD image node. */}} {{- $url := partial "seo/lib/absolute-url.html" (dict "value" $str "key" "seo.image") -}} {{- $type := "" -}} {{- $lower := lower $str -}} {{- if hasSuffix $lower ".png" -}} {{- $type = "image/png" -}} {{- else if or (hasSuffix $lower ".jpg") (hasSuffix $lower ".jpeg") -}} {{- $type = "image/jpeg" -}} {{- else if hasSuffix $lower ".webp" -}} {{- $type = "image/webp" -}} {{- else if hasSuffix $lower ".gif" -}} {{- $type = "image/gif" -}} {{- else if hasSuffix $lower ".svg" -}} {{- $type = "image/svg+xml" -}} {{- end -}} {{- $result = dict "url" $url "alt" $alt "natural" (dict "url" $url) -}} {{- with $type -}} {{- $result = merge $result (dict "type" .) -}} {{- end -}} {{- end -}} {{/* Final alt fallbacks apply only when an image actually resolved. */}} {{- if $result.url -}} {{- if not $result.alt -}} {{- $altFallback := "" -}} {{- with $page.Param "seo.image_alt" -}} {{- $altFallback = . -}} {{- else -}} {{- $t := partial "seo/resolve/title.html" $page -}} {{- $altFallback = $t.title -}} {{- end -}} {{- $result = merge $result (dict "alt" $altFallback) -}} {{- end -}} {{- end -}} {{- return $result -}}
