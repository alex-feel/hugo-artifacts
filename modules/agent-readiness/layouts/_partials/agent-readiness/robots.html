{{/* agent-readiness/robots.html -- renderer (RENDERER tier). Purpose: emit the whole robots.txt body -- the catch-all group carrying the Content-Signal declaration, one group per configured AI-crawler token resolved through the shipped registry, any verbatim extra lines, and the Sitemap directive. Defaults are permissive BY CONSTRUCTION: with no configuration at all the output is a directive-free `User-agent: *` group and the Sitemap line, which RFC 9309 reads as fully permissive. No `Allow: /` ships either: an Allow and a Disallow of equal path length tie, RFC 9309 section 2.2.2 resolves the tie in favor of Allow, so a shipped `Allow: /` would sit in every group and silently neutralize the `Disallow: /` a consumer configures to block a crawler. Every restriction is opt-in, because a Disallow shipped as a module default would deindex a consumer's site on the build after they imported the module. The document is assembled as a slice of lines and emitted once, rather than written inline with whitespace-trimming markers. robots.txt is a line-oriented format whose group separation is blank-line significant, and line assembly is the only way to keep that exact and reviewable across every combination of configured and unconfigured keys -- the inline form emitted a doubled blank line in the commonest case of all, a site with no bot groups and no extra lines. Because the format is line-oriented, every consumer-authored value is collapsed to one line through lib/inline.html before it joins its directive: a path value carrying an embedded line break would otherwise inject a directive line of its own. The `extra` lines are the deliberate exception -- that key exists to carry verbatim robots.txt lines, so its entries pass through untouched, embedded line breaks included. Expected context (dict): - .page (Page) - The home page (robots.txt renders from it). - .cfg (dict) - The resolved config from agent-readiness/config.html, resolved with site_only: true (every robots key is SITE-SCOPED). Return value: none. This partial renders. */}} {{- $cfg := .cfg -}} {{- $robots := $cfg.robots -}} {{- $data := index hugo.Data "agent-readiness" | default dict -}} {{- $registry := index $data "bots" | default dict -}} {{- if and $cfg.enable $robots.enable -}} {{- $lines := slice -}} {{/* The catch-all group. Content-Signal sits INSIDE it, per the contentsignals.org convention. */}} {{- $lines = $lines | append "User-agent: *" -}} {{- with $robots.content_signal -}} {{- $lines = $lines | append (printf "Content-Signal: %s" (partial "agent-readiness/lib/inline.html" .)) -}} {{- end -}} {{- range $robots.allow | default slice -}} {{- $lines = $lines | append (printf "Allow: %s" (partial "agent-readiness/lib/inline.html" .)) -}} {{- end -}} {{- range $robots.disallow | default slice -}} {{- $lines = $lines | append (printf "Disallow: %s" (partial "agent-readiness/lib/inline.html" .)) -}} {{- end -}} {{/* One group per configured token, each separated by exactly one blank line. An unknown key is skipped with one deduplicated warning, never emitted as a literal: a registry key in a User-agent line matches no crawler and silently does nothing, which is strictly worse than the group being absent. */}} {{- range $key := $robots.bots | default slice -}} {{- $k := lower (trim (printf "%v" $key) " \t\n\r") -}} {{- $entry := index $registry $k -}} {{- $token := "" -}} {{- if reflect.IsMap $entry -}}{{- $token = partial "agent-readiness/lib/inline.html" ($entry.token | default "") -}}{{- end -}} {{- if $token -}} {{- $lines = $lines | append "" | append (printf "User-agent: %s" $token) -}} {{- range $robots.bots_allow | default slice -}} {{- $lines = $lines | append (printf "Allow: %s" (partial "agent-readiness/lib/inline.html" .)) -}} {{- end -}} {{- range $robots.bots_disallow | default slice -}} {{- $lines = $lines | append (printf "Disallow: %s" (partial "agent-readiness/lib/inline.html" .)) -}} {{- end -}} {{- else -}} {{- partial "agent-readiness/lib/warn.html" (dict "key" (printf "agent-readiness:warn:unknown-bot:%s" $k) "message" (printf "[agent-readiness] Ignoring the unknown robots bot key %q: no such entry in data/agent-readiness/bots.toml, and emitting the key itself as a User-agent token would match no crawler. Use one of the registry keys, or add your own entry by shadowing that data file in your site." $k)) -}} {{- end -}} {{- end -}} {{/* Verbatim extra lines, so a consumer can express any directive this module does not model. */}} {{- with $robots.extra -}} {{- $lines = $lines | append "" -}} {{- range . -}} {{- $lines = $lines | append (printf "%v" .) -}} {{- end -}} {{- end -}} {{/* Sitemap, built with absURL rather than a hardcoded absolute URL, so development and production builds agree and a subpath baseURL is honored. */}} {{- if $robots.sitemap -}} {{- $lines = $lines | append "" | append (printf "Sitemap: %s" ("sitemap.xml" | absURL)) -}} {{- end -}} {{/* safeHTML because this partial lives in the .html template namespace while robots.txt is a plain-text document: without it a directive containing an ampersand would be HTML-escaped into an entity. The trailing newline is deliberate -- a text file ends with one. */}} {{- printf "%s\n" (delimit $lines "\n") | safeHTML -}} {{- end -}}
