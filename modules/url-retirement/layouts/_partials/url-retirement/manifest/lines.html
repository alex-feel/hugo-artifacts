{{/* url-retirement/manifest/lines.html -- INTERNAL (document tier). Purpose: builds every line of /url-manifest.txt for one language -- a `#` comment header, then every URL the build publishes, sorted, one per line -- and RETURNS them. It prints nothing; see redirects/lines.html for why a partial in this module never emits into a plain-text document. What it lists: every page of this language crossed with its output formats (so an RSS feed, a Markdown twin and a JSON representation each count as the published URLs they are), every pager URL the site registered through register-pagers.html, and whatever url_retirement.manifest.extra names. What it cannot list, and says so in its own header: files copied verbatim from static/, which Hugo exposes to no template, and the site-level documents whose publication is decided by settings no template can read (sitemap.xml, robots.txt, 404.html). `extra` exists for exactly those. One caveat worth knowing rather than hiding: .OutputFormats reports the formats CONFIGURED for a page, and a format with no matching template publishes nothing while still appearing there. Hugo warns loudly for that case ("found no layout file for ..."), so the condition is visible at build time rather than only here. Context (dict): - .page (Page) The home page rendering the document. Return value: slice of strings, one line each. Empty when the module or the document is switched off. One terminal return. */}} {{- $page := .page -}} {{- $cfg := partial "url-retirement/config.html" (dict "page" $page) -}} {{- $lines := slice -}} {{- if and $cfg.enable $cfg.manifest.enable -}} {{/* Every URL is normalized through lib/prefix-url.html so a subpath site and a canonified one publish the same manifest: under canonifyURLs, .RelPermalink stops carrying the baseURL's path segment. */}} {{- $base := strings.TrimSuffix "/" (urls.Parse $page.Site.BaseURL).Path -}} {{- $urls := slice -}} {{- range $p := $page.Site.Pages -}} {{- if $cfg.manifest.output_formats -}} {{- range $f := $p.OutputFormats -}} {{- with $f.RelPermalink -}} {{- $urls = $urls | append (partial "url-retirement/lib/prefix-url.html" (dict "base" $base "url" .)) -}} {{- end -}} {{- end -}} {{- else -}} {{- with $p.RelPermalink -}} {{- $urls = $urls | append (partial "url-retirement/lib/prefix-url.html" (dict "base" $base "url" .)) -}} {{- end -}} {{- end -}} {{- end -}} {{/* Pagers, recorded during the html pass by the site's own list templates. The store key carries the language because the store is build-global: without it, the second language's manifest would republish the first language's pager URLs as its own. */}} {{- range hugo.Store.GetSortedMapValues (printf "url-retirement:pagers:%s" $cfg.lang) -}} {{- $urls = $urls | append (partial "url-retirement/lib/prefix-url.html" (dict "base" $base "url" (printf "%v" .))) -}} {{- end -}} {{- range $cfg.manifest.extra -}} {{- $urls = $urls | append (partial "url-retirement/lib/prefix-url.html" (dict "base" $base "url" .)) -}} {{- end -}} {{- $sorted := slice -}} {{- with $urls -}} {{- $sorted = uniq (sort .) -}} {{- end -}} {{/* A sibling is named only when that language will actually publish one. Two independent things decide that: the format has to be wired onto that language's home page, and that language's own configuration has to leave the manifest switched on -- a per-language `manifest.enable = false` is invisible in the format wiring. Naming a file that was never written breaks the one discovery mechanism this header exists to provide, so both are checked rather than the cheaper one. */}} {{- $siblings := slice -}} {{- range $s := hugo.Sites -}} {{- if ne $s.Language.Name $cfg.lang -}} {{- with $s.Home.OutputFormats.Get "urlmanifest" -}} {{- $scfg := partial "url-retirement/config.html" (dict "page" $s.Home) -}} {{- if and $scfg.enable $scfg.manifest.enable -}} {{- $siblings = $siblings | append (partial "url-retirement/lib/prefix-url.html" (dict "base" $base "url" .RelPermalink)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $lines = slice (printf "# Every URL this build publishes for language %q, sorted, one per line." $cfg.lang) "# Generated by the url-retirement Hugo module. Lines starting with # are comments." "# Not listed: files copied verbatim from static/, which Hugo exposes to no" "# template, and any document whose publication is decided by a setting no" "# template can read. Name those in url_retirement.manifest.extra." -}} {{- with $siblings -}} {{- $lines = $lines | append (printf "# Other languages: %s" (delimit . ", ")) -}} {{- end -}} {{- $lines = $lines | append (printf "# %d URLs follow." (len $sorted)) -}} {{- with $sorted -}} {{- $lines = $lines | append . -}} {{- end -}} {{- end -}} {{- return $lines -}}
