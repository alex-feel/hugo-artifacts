{{/* og-image/resolve/overlays.html -- INTERNAL (resolver tier). Purpose: turns a card template's `overlay` array into images.Overlay filters. Overlays are composited in declaration order and UNDER the text, so a logo can never hide the words the card exists to show. Each entry names its image one of four ways -- an `asset` path, a `param` naming a front-matter key that holds a path, a `resource` glob matched against the page's own bundle, or a `data` key naming a path in a data file -- and og-image/lib/resolve-image.html owns that vocabulary for the module, so an overlay and a background name their artwork identically. What is decided HERE is what each outcome MEANS to an overlay: the split between a warning and silence follows WHO made the mistake, not which source it happened under. An entry whose own `src`, `key` or `match` is missing, or whose `match` is not a pattern that can be matched at all, is a card-template mistake no page can satisfy and warns once; a param or bundle resource the page simply does not have is the ordinary case and stays silent. `data` has no page in it -- the key and the value are both the site's own -- so every data fault is a mistake and warns once. Either way the rest of the card still renders. Placement: images.Overlay takes absolute pixel coordinates from the top-left corner, so the configured `anchor` plus the overlay's own size decide the base coordinate and `x`/`y` move it INWARD from that anchor -- a left or top anchor adds them, a right or bottom anchor subtracts them, and a centered axis offsets from the center. The overlay is resized before its position is computed, because the position depends on the size that will be drawn. Context (dict): - .page (Page) The current page. - .opts (dict) The resolved options from og-image/config.html. - .name (string) The card template's name. - .template (map) The card template's table. Return value: dict{filters (slice), any (bool)}. One terminal return. */}} {{- $page := .page -}} {{- $opts := .opts -}} {{- $name := .name -}} {{- $tpl := .template -}} {{- $filters := slice -}} {{- $any := false -}} {{- $entries := partial "og-image/lib/as-map-list.html" (dict "value" (index $tpl "overlay") "key" (printf "og-image:warn:bad-overlay-entry:%s" $name) "label" (printf "the overlays of card template %q" $name)) -}} {{- range $i, $ov := $entries -}} {{- $found := partial "og-image/lib/resolve-image.html" (dict "page" $page "spec" $ov "default" "asset") -}} {{- $res := $found.resource -}} {{- $label := $found.label -}} {{/* An `absent` outcome is the page not carrying the image, which is the design and stays silent. Every other outcome is a mistake somebody made, named in the terms of the tier that made it. */}} {{- if eq $found.reason "unknown-source" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unknown-enum:overlay-source:%s" $found.source) "message" (printf "[og-image] Unknown overlay source %q in card template %q; that overlay is dropped (allowed: %s)." $found.source $name (delimit $found.known ", "))) -}} {{- else if eq $found.reason "no-locator" -}} {{- if eq $found.source "asset" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-no-src:%s:%d" $name $i) "message" (printf "[og-image] Overlay %d of card template %q names no image: an overlay reading from assets/ needs a src path. That overlay is dropped." $i $name)) -}} {{- else if eq $found.source "param" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-no-key:%s:%d" $name $i) "message" (printf "[og-image] Overlay %d of card template %q names no image: an overlay reading from a page parameter needs a key naming that parameter. That overlay is dropped." $i $name)) -}} {{- else if eq $found.source "data" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-data-no-key:%s:%d" $name $i) "message" (printf "[og-image] Overlay %d of card template %q names no image: an overlay reading from a data file needs a key naming the value that holds its path. That overlay is dropped." $i $name)) -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-no-match:%s:%d" $name $i) "message" (printf "[og-image] Overlay %d of card template %q names no image: an overlay reading from the page's own resources needs a match pattern. That overlay is dropped." $i $name)) -}} {{- end -}} {{- else if eq $found.reason "not-found" -}} {{/* The key carries the source as well as the path, because the wording differs by source and the first message stored under a key wins: one path named by both an asset src and a data value must not have one spelling suppress the other's provenance. */}} {{- if eq $found.source "data" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:missing-overlay:%s:%s" $found.source $label) "message" (printf "[og-image] The overlay %q named in data/ by card template %q was not found in the page's resources or under assets/; that overlay is dropped and the card is composed without it." $label $name)) -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:missing-overlay:%s:%s" $found.source $label) "message" (printf "[og-image] The overlay %q used by card template %q was not found under assets/; that overlay is dropped and the card is composed without it." $label $name)) -}} {{- end -}} {{- else if eq $found.reason "no-value" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-data-no-value:%s:%d" $name $i) "message" (printf "[og-image] Overlay %d of card template %q reads the data key %q, which names no value in data/; that overlay is dropped." $i $name $label)) -}} {{- else if eq $found.reason "bad-value" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-data-bad-value:%s:%d" $name $i) "message" (printf "[og-image] Overlay %d of card template %q reads the data key %q, which holds a table or a list rather than a path; that overlay is dropped." $i $name $label)) -}} {{- else if eq $found.reason "bad-match" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-bad-match:%s:%s" $name $label) "message" (printf "[og-image] The overlay pattern %q in card template %q is not a pattern Hugo can match against a page's resources, so it names no image on any page and that overlay is dropped. An unclosed [ is the usual cause." $label $name)) -}} {{- else if eq $found.reason "unusable" -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unusable-overlay:%s" $label) "message" (printf "[og-image] The overlay %q used by card template %q carries no image Hugo can process (an SVG, an unsupported format, or a file whose bytes are not the image its extension claims); that overlay is dropped." $label $name)) -}} {{- end -}} {{- if $res -}} {{/* Resize first: the anchor arithmetic below needs the size that will actually be drawn. */}} {{- $ok := true -}} {{- if isset $ov "width" -}} {{- $n := partial "og-image/lib/int.html" (index $ov "width") -}} {{- if and $n.ok (ge $n.value 1) (le $n.value 100000) -}} {{- $sized := try ($res.Resize (printf "%dx" $n.value)) -}} {{- with $sized.Err -}} {{- $ok = false -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unresizable-overlay:%s" $label) "message" (printf "[og-image] Could not resize the overlay %q used by card template %q: %s. That overlay is dropped." $label $name .)) -}} {{- else -}} {{- $res = $sized.Value -}} {{- end -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:bad-int:%s:overlay%d:width:%v" $name $i (index $ov "width")) "message" (printf "[og-image] Ignoring the width value %q on overlay %d of card template %q (expected a whole number of pixels); the overlay is drawn at its own size." (printf "%v" (index $ov "width")) $i $name)) -}} {{- end -}} {{- end -}} {{- if $ok -}} {{- $opacity := 1.0 -}} {{- if isset $ov "opacity" -}} {{/* The pattern bounds the SHAPE but not the magnitude, and `float` RAISES on a digit string too long for a float64, so the cast is caught rather than trusted. A shape that does not match, a cast that fails and a number outside 0..1 are one mistake to the site -- the value is not an opacity -- so they share the sentinel below and warn once in one place. */}} {{- $raw := trim (printf "%v" (index $ov "opacity")) " \t\n\r" -}} {{- $f := -1.0 -}} {{- if findRE `^\d+(\.\d+)?$` $raw -}} {{- $parsed := try (float $raw) -}} {{- if not $parsed.Err -}} {{- $f = $parsed.Value -}} {{- end -}} {{- end -}} {{- if and (ge $f 0.0) (le $f 1.0) -}} {{- $opacity = $f -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:bad-float:%s:overlay%d:opacity:%s" $name $i $raw) "message" (printf "[og-image] Ignoring the opacity value %q on overlay %d of card template %q (expected a number from 0 to 1); the overlay is drawn fully opaque." $raw $i $name)) -}} {{- end -}} {{- end -}} {{- if lt $opacity 1.0 -}} {{- $faded := try ($res.Filter (images.Opacity $opacity)) -}} {{- with $faded.Err -}} {{- $ok = false -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unfadeable-overlay:%s" $label) "message" (printf "[og-image] Could not apply opacity to the overlay %q used by card template %q: %s. That overlay is dropped." $label $name .)) -}} {{- else -}} {{- $res = $faded.Value -}} {{- end -}} {{- end -}} {{- end -}} {{- if $ok -}} {{- $anchor := lower (trim (printf "%v" (index $ov "anchor" | default "topleft")) " \t\n\r") -}} {{- if not (in (slice "topleft" "top" "topright" "left" "center" "right" "bottomleft" "bottom" "bottomright") $anchor) -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unknown-enum:overlay-anchor:%s" $anchor) "message" (printf "[og-image] Unknown overlay anchor %q in card template %q; using \"topleft\" instead (allowed: topleft, top, topright, left, center, right, bottomleft, bottom, bottomright)." $anchor $name)) -}} {{- $anchor = "topleft" -}} {{- end -}} {{- $dx := 0 -}} {{- if isset $ov "x" -}} {{- $n := partial "og-image/lib/int.html" (index $ov "x") -}} {{- if $n.ok -}}{{- $dx = $n.value -}}{{- end -}} {{- end -}} {{- $dy := 0 -}} {{- if isset $ov "y" -}} {{- $n := partial "og-image/lib/int.html" (index $ov "y") -}} {{- if $n.ok -}}{{- $dy = $n.value -}}{{- end -}} {{- end -}} {{- $freeX := sub $opts.width $res.Width -}} {{- $freeY := sub $opts.height $res.Height -}} {{- $x := $dx -}} {{- if in (slice "top" "center" "bottom") $anchor -}} {{- $x = add (int (math.Floor (div (float $freeX) 2.0))) $dx -}} {{- else if in (slice "topright" "right" "bottomright") $anchor -}} {{- $x = sub $freeX $dx -}} {{- end -}} {{- $y := $dy -}} {{- if in (slice "left" "center" "right") $anchor -}} {{- $y = add (int (math.Floor (div (float $freeY) 2.0))) $dy -}} {{- else if in (slice "bottomleft" "bottom" "bottomright") $anchor -}} {{- $y = sub $freeY $dy -}} {{- end -}} {{- $filters = $filters | append (images.Overlay $res $x $y) -}} {{- $any = true -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "filters" $filters "any" $any) -}}
