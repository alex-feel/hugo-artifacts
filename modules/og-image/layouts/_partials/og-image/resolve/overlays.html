{{/* og-image/resolve/overlays.html -- INTERNAL (resolver tier). Purpose: turns a card template's `overlay` array into images.Overlay filters. Overlays are composited in declaration order and UNDER the text, so a logo can never hide the words the card exists to show. Each entry names its image one of three ways: an `asset` path (the shared badge every card carries), a `param` naming a front-matter key that holds a path, or a `resource` glob matched against the page's own bundle (the per-page avatar). A shared asset that names nothing is a configuration mistake and warns once; a per-page param or bundle resource the page simply does not have is the ordinary case and stays silent. Either way the rest of the card still renders. Placement: images.Overlay takes absolute pixel coordinates from the top-left corner, so the configured `anchor` plus the overlay's own size decide the base coordinate and `x`/`y` move it INWARD from that anchor -- a left or top anchor adds them, a right or bottom anchor subtracts them, and a centered axis offsets from the center. The overlay is resized before its position is computed, because the position depends on the size that will be drawn. Context (dict): - .page (Page) The current page. - .opts (dict) The resolved options from og-image/config.html. - .name (string) The card template's name. - .template (map) The card template's table. Return value: dict{filters (slice), any (bool)}. One terminal return. */}} {{- $page := .page -}} {{- $opts := .opts -}} {{- $name := .name -}} {{- $tpl := .template -}} {{- $filters := slice -}} {{- $any := false -}} {{- $entries := partial "og-image/lib/as-map-list.html" (dict "value" (index $tpl "overlay") "key" (printf "og-image:warn:bad-overlay-entry:%s" $name) "label" (printf "the overlays of card template %q" $name)) -}} {{- range $i, $ov := $entries -}} {{- $source := lower (trim (printf "%v" (index $ov "source" | default "asset")) " \t\n\r") -}} {{- $res := "" -}} {{- $label := "" -}} {{- if eq $source "asset" -}} {{- $src := trim (printf "%v" (index $ov "src" | default "")) " \t\n\r" -}} {{- $label = $src -}} {{- if $src -}} {{- with resources.Get $src -}} {{- $res = . -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:missing-overlay:%s" $src) "message" (printf "[og-image] The overlay %q used by card template %q was not found under assets/; that overlay is dropped and the card is composed without it." $src $name)) -}} {{- end -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:overlay-no-src:%s:%d" $name $i) "message" (printf "[og-image] Overlay %d of card template %q names no image: an overlay reading from assets/ needs a src path. That overlay is dropped." $i $name)) -}} {{- end -}} {{- else if eq $source "param" -}} {{/* A page that simply does not carry the parameter is the design, not a mistake, so it is skipped in silence. */}} {{- $key := trim (printf "%v" (index $ov "key" | default "")) " \t\n\r" -}} {{- if $key -}} {{- $v := $page.Param $key -}} {{- if and (ne $v nil) (not (reflect.IsMap $v)) (not (reflect.IsSlice $v)) -}} {{- $p := trim (printf "%v" $v) " \t\n\r" -}} {{- $label = $p -}} {{- if $p -}} {{- with $page.Resources.GetMatch $p -}} {{- $res = . -}} {{- else -}} {{- with resources.Get $p -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- else if eq $source "resource" -}} {{- $match := trim (printf "%v" (index $ov "match" | default "")) " \t\n\r" -}} {{- $label = $match -}} {{- if $match -}} {{- with $page.Resources.GetMatch $match -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unknown-enum:overlay-source:%s" $source) "message" (printf "[og-image] Unknown overlay source %q in card template %q; that overlay is dropped (allowed: asset, param, resource)." $source $name)) -}} {{- end -}} {{- if $res -}} {{/* A resource can be media-typed as an image and still carry nothing Hugo can decode, and reading .Width is what finds out. */}} {{- $usable := eq $res.ResourceType "image" -}} {{- if and $usable (eq $res.MediaType.SubType "svg") -}} {{- $usable = false -}} {{- end -}} {{- if $usable -}} {{- $probe := try $res.Width -}} {{- $usable = and (not $probe.Err) (gt (int ($probe.Value | default 0)) 0) -}} {{- end -}} {{- if not $usable -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unusable-overlay:%s" $label) "message" (printf "[og-image] The overlay %q used by card template %q carries no image Hugo can process (an SVG, an unsupported format, or a file whose bytes are not the image its extension claims); that overlay is dropped." $label $name)) -}} {{- else -}} {{/* Resize first: the anchor arithmetic below needs the size that will actually be drawn. */}} {{- $ok := true -}} {{- if isset $ov "width" -}} {{- $n := partial "og-image/lib/int.html" (index $ov "width") -}} {{- if and $n.ok (ge $n.value 1) (le $n.value 100000) -}} {{- $sized := try ($res.Resize (printf "%dx" $n.value)) -}} {{- with $sized.Err -}} {{- $ok = false -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unresizable-overlay:%s" $label) "message" (printf "[og-image] Could not resize the overlay %q used by card template %q: %s. That overlay is dropped." $label $name .)) -}} {{- else -}} {{- $res = $sized.Value -}} {{- end -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:bad-int:%s:overlay%d:width:%v" $name $i (index $ov "width")) "message" (printf "[og-image] Ignoring the width value %q on overlay %d of card template %q (expected a whole number of pixels); the overlay is drawn at its own size." (printf "%v" (index $ov "width")) $i $name)) -}} {{- end -}} {{- end -}} {{- if $ok -}} {{- $opacity := 1.0 -}} {{- if isset $ov "opacity" -}} {{- $raw := trim (printf "%v" (index $ov "opacity")) " \t\n\r" -}} {{- if findRE `^\d+(\.\d+)?$` $raw -}} {{- $f := float $raw -}} {{- if and (ge $f 0.0) (le $f 1.0) -}} {{- $opacity = $f -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:bad-float:%s:overlay%d:opacity:%s" $name $i $raw) "message" (printf "[og-image] Ignoring the opacity value %q on overlay %d of card template %q (expected a number from 0 to 1); the overlay is drawn fully opaque." $raw $i $name)) -}} {{- end -}} {{- else -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:bad-float:%s:overlay%d:opacity:%s" $name $i $raw) "message" (printf "[og-image] Ignoring the opacity value %q on overlay %d of card template %q (expected a number from 0 to 1); the overlay is drawn fully opaque." $raw $i $name)) -}} {{- end -}} {{- end -}} {{- if lt $opacity 1.0 -}} {{- $faded := try ($res.Filter (images.Opacity $opacity)) -}} {{- with $faded.Err -}} {{- $ok = false -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unfadeable-overlay:%s" $label) "message" (printf "[og-image] Could not apply opacity to the overlay %q used by card template %q: %s. That overlay is dropped." $label $name .)) -}} {{- else -}} {{- $res = $faded.Value -}} {{- end -}} {{- end -}} {{- end -}} {{- if $ok -}} {{- $anchor := lower (trim (printf "%v" (index $ov "anchor" | default "topleft")) " \t\n\r") -}} {{- if not (in (slice "topleft" "top" "topright" "left" "center" "right" "bottomleft" "bottom" "bottomright") $anchor) -}} {{- partial "og-image/lib/warn.html" (dict "key" (printf "og-image:warn:unknown-enum:overlay-anchor:%s" $anchor) "message" (printf "[og-image] Unknown overlay anchor %q in card template %q; using \"topleft\" instead (allowed: topleft, top, topright, left, center, right, bottomleft, bottom, bottomright)." $anchor $name)) -}} {{- $anchor = "topleft" -}} {{- end -}} {{- $dx := 0 -}} {{- if isset $ov "x" -}} {{- $n := partial "og-image/lib/int.html" (index $ov "x") -}} {{- if $n.ok -}}{{- $dx = $n.value -}}{{- end -}} {{- end -}} {{- $dy := 0 -}} {{- if isset $ov "y" -}} {{- $n := partial "og-image/lib/int.html" (index $ov "y") -}} {{- if $n.ok -}}{{- $dy = $n.value -}}{{- end -}} {{- end -}} {{- $freeX := sub $opts.width $res.Width -}} {{- $freeY := sub $opts.height $res.Height -}} {{- $x := $dx -}} {{- if in (slice "top" "center" "bottom") $anchor -}} {{- $x = add (int (math.Floor (div (float $freeX) 2.0))) $dx -}} {{- else if in (slice "topright" "right" "bottomright") $anchor -}} {{- $x = sub $freeX $dx -}} {{- end -}} {{- $y := $dy -}} {{- if in (slice "left" "center" "right") $anchor -}} {{- $y = add (int (math.Floor (div (float $freeY) 2.0))) $dy -}} {{- else if in (slice "bottomleft" "bottom" "bottomright") $anchor -}} {{- $y = sub $freeY $dy -}} {{- end -}} {{- $filters = $filters | append (images.Overlay $res $x $y) -}} {{- $any = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "filters" $filters "any" $any) -}}
