{{/* Renders a GitHub repository reference in one of five display variants. Parameters: - url (string, required): Full GitHub repository URL - variant (string, optional): Display variant. One of: inline, card, stats, lang, hero. Default: "card" - name (string, optional): Display name override (defaults to repo name from URL/API) - description (string, optional): Description text override (defaults to API description) - class (string, optional): Additional CSS class(es) appended to the root element - updated (string, optional): Display mode for the last-push time label in the card and hero variants. One of: relative, date, none. Default: "relative". "relative" renders a calendar-day-granularity phrase ("today", "yesterday", "5 days ago") that stays accurate while the site rebuilds at least daily; "date" renders the localized absolute date, which never goes stale at any rebuild cadence; "none" omits the element entirely. Usage: github-repo url="https://github.com/owner/repo" github-repo url="https://github.com/owner/repo" variant="hero" github-repo url="https://github.com/owner/repo" variant="inline" name="My Fork" class="featured" github-repo url="https://github.com/owner/repo" updated="date" */}} {{- $url := .Get "url" -}} {{- $variant := .Get "variant" | default "card" -}} {{- $name := .Get "name" | default "" -}} {{- $description := .Get "description" | default "" -}} {{- $class := .Get "class" | default "" -}} {{- $updated := .Get "updated" | default "relative" -}} {{/* Validate required url parameter */}} {{- if not $url -}} {{- errorf "The %q shortcode requires a 'url' parameter. See %s" .Name .Position -}} {{- end -}} {{/* Validate variant against allowed values */}} {{- $validVariants := slice "inline" "card" "stats" "lang" "hero" -}} {{- if not (in $validVariants $variant) -}} {{- errorf "The %q shortcode received invalid variant %q. Must be one of: inline, card, stats, lang, hero. See %s" .Name $variant .Position -}} {{- end -}} {{/* Validate the last-push label display mode against allowed values */}} {{- $validUpdated := slice "relative" "date" "none" -}} {{- if not (in $validUpdated $updated) -}} {{- errorf "The %q shortcode received invalid updated %q. Must be one of: relative, date, none. See %s" .Name $updated .Position -}} {{- end -}} {{/* Fetch and normalize repository data */}} {{- $ctx := dict "url" $url "variant" $variant "name" $name "description" $description "position" .Position -}} {{- $data := partial "github-repo/fetch.html" $ctx -}} {{/* Merge the per-call presentation settings into the fetched data */}} {{- $data = merge $data (dict "extraClass" $class "updatedMode" $updated) -}} {{/* Dispatch to variant partial */}} {{- partial (printf "github-repo/%s.html" $variant) $data -}}
